/* jump.js - redirect on include, target is others.aipages.dev
 * Usage: <script src="https://your-domain/jump.js"></script>
 * To change the target: edit the TARGET line only.
 */
(function () {
  'use strict';

  // Redirect target (change this line only)
  var TARGET = 'https://others.aipages.dev/';

  // Get referring host: prefer document.referrer, fallback to current hostname
  function hostOf(u) {
    try { return new URL(u).hostname || ''; } catch (e) { return ''; }
  }

  var ref = hostOf(document.referrer) || location.hostname || '';

  // Do not redirect when already on the target site (avoid loop)
  if (/^others\.aipages\.dev$/i.test(location.hostname)) return;

  // Do not redirect twice
  if (window.__jump_done__) return;
  window.__jump_done__ = true;

  location.replace(TARGET + '?' + encodeURIComponent(ref));
})();
